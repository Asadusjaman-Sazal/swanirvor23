-- =========================================================================
-- 1. Create the members table (relates to Supabase Auth Users)
-- =========================================================================
CREATE TABLE IF NOT EXISTS public.members (
    id SERIAL PRIMARY KEY,
    name TEXT NOT NULL,
    email TEXT NOT NULL UNIQUE,
    avatar_url TEXT,
    total_savings DOUBLE PRECISION DEFAULT 0.0,
    role TEXT DEFAULT 'Member',
    status TEXT DEFAULT 'Active',
    received_admin_notification BOOLEAN DEFAULT false,
    password TEXT DEFAULT 'password',
    membership_no TEXT DEFAULT 'LS-2023-000',
    user_id UUID REFERENCES auth.users(id) ON DELETE CASCADE DEFAULT auth.uid(),
    created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- =========================================================================
-- 2. Create the savings table
-- =========================================================================
CREATE TABLE IF NOT EXISTS public.savings (
    id SERIAL PRIMARY KEY,
    member_id INTEGER REFERENCES public.members(id) ON DELETE CASCADE,
    member_name TEXT NOT NULL,
    amount DOUBLE PRECISION NOT NULL,
    date_text TEXT NOT NULL,
    timestamp BIGINT NOT NULL,
    deleted BOOLEAN DEFAULT false,
    deleted_at TIMESTAMP WITH TIME ZONE,
    user_id UUID REFERENCES auth.users(id) ON DELETE CASCADE DEFAULT auth.uid(),
    created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- Soft-delete columns for pre-existing savings tables (idempotent)
ALTER TABLE public.savings ADD COLUMN IF NOT EXISTS deleted BOOLEAN DEFAULT false;
ALTER TABLE public.savings ADD COLUMN IF NOT EXISTS deleted_at TIMESTAMP WITH TIME ZONE;
CREATE INDEX IF NOT EXISTS idx_savings_deleted ON public.savings(deleted);

-- =========================================================================
-- 3. Create the app_settings table
-- =========================================================================
CREATE TABLE IF NOT EXISTS public.app_settings (
    id SERIAL PRIMARY KEY,
    personal_goal DOUBLE PRECISION DEFAULT 500.0,
    profile_name TEXT NOT NULL DEFAULT 'John Doe',
    membership_no TEXT NOT NULL DEFAULT 'LS-2023-089',
    is_dark_mode BOOLEAN DEFAULT false,
    notification_day TEXT DEFAULT 'Thursday',
    notification_time TEXT DEFAULT '09:00',
    notification_text TEXT DEFAULT 'Reminder: Your weekly contribution of ৳500 is due tomorrow.',
    enable_notifications BOOLEAN DEFAULT true,
    profile_image_url TEXT,
    user_id UUID REFERENCES auth.users(id) ON DELETE CASCADE DEFAULT auth.uid() UNIQUE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- =========================================================================
-- 4. Create the change_requests table
-- =========================================================================
CREATE TABLE IF NOT EXISTS public.change_requests (
    id SERIAL PRIMARY KEY,
    savings_id INTEGER,
    member_id INTEGER REFERENCES public.members(id) ON DELETE CASCADE,
    member_name TEXT NOT NULL,
    request_type TEXT NOT NULL,
    original_amount DOUBLE PRECISION NOT NULL,
    new_amount DOUBLE PRECISION,
    original_date_text TEXT NOT NULL,
    new_date_text TEXT,
    status TEXT DEFAULT 'Pending',
    timestamp BIGINT NOT NULL,
    user_id UUID REFERENCES auth.users(id) ON DELETE CASCADE DEFAULT auth.uid(),
    created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- =========================================================================
-- 5. Enable Row-Level Security (RLS) on all tables
-- =========================================================================
ALTER TABLE public.members ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.savings ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.app_settings ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.change_requests ENABLE ROW LEVEL SECURITY;

-- =========================================================================
-- 6. Define Security Policies using auth.uid()
-- =========================================================================

-- Members Table Policies
CREATE POLICY "Allow read for all authenticated users" ON public.members
    FOR SELECT TO authenticated USING (true);

CREATE POLICY "Allow insert/update for own profile" ON public.members
    FOR ALL TO authenticated USING (user_id = auth.uid()) WITH CHECK (user_id = auth.uid());

CREATE POLICY "Admins can update all members" ON public.members
    FOR UPDATE TO authenticated USING (
        EXISTS (
            SELECT 1 FROM public.members 
            WHERE user_id = auth.uid() AND role = 'Admin'
        )
    );

CREATE POLICY "Admins can delete all members" ON public.members
    FOR DELETE TO authenticated USING (
        EXISTS (
            SELECT 1 FROM public.members 
            WHERE user_id = auth.uid() AND role = 'Admin'
        )
    );

-- Savings Table Policies
CREATE POLICY "Allow read for all authenticated users" ON public.savings
    FOR SELECT TO authenticated USING (true);

CREATE POLICY "Users can manage own savings, Admins can manage all" ON public.savings
    FOR ALL TO authenticated USING (
        user_id = auth.uid() OR 
        EXISTS (
            SELECT 1 FROM public.members 
            WHERE user_id = auth.uid() AND role = 'Admin'
        )
    ) WITH CHECK (
        user_id = auth.uid() OR 
        EXISTS (
            SELECT 1 FROM public.members 
            WHERE user_id = auth.uid() AND role = 'Admin'
        )
    );

-- App Settings Policies
CREATE POLICY "Users can manage own settings" ON public.app_settings
    FOR ALL TO authenticated USING (user_id = auth.uid()) WITH CHECK (user_id = auth.uid());

-- Change Requests Policies
CREATE POLICY "Users can read own change requests, Admins can read all" ON public.change_requests
    FOR SELECT TO authenticated USING (
        user_id = auth.uid() OR 
        EXISTS (
            SELECT 1 FROM public.members 
            WHERE user_id = auth.uid() AND role = 'Admin'
        )
    );

CREATE POLICY "Users can create own change requests, Admins can manage all" ON public.change_requests
    FOR ALL TO authenticated USING (
        user_id = auth.uid() OR 
        EXISTS (
            SELECT 1 FROM public.members 
            WHERE user_id = auth.uid() AND role = 'Admin'
        )
    ) WITH CHECK (
        user_id = auth.uid() OR 
        EXISTS (
            SELECT 1 FROM public.members 
            WHERE user_id = auth.uid() AND role = 'Admin'
        )
    );

-- ====================================================================
-- RLS Policy for avatar Storage Bucket
-- ====================================================================
-- 1. Policy for SELECT (Allows anyone to view any profile picture in the avatars bucket)
CREATE POLICY "Allow public read access to avatars"
ON storage.objects FOR SELECT
USING ( bucket_id = 'avatars' );

-- 2. Policy for INSERT (Allows logged-in users to upload only their own avatar files)
CREATE POLICY "Allow authenticated users to upload their own avatars"
ON storage.objects FOR INSERT
TO authenticated
WITH CHECK (
  bucket_id = 'avatars' 
  AND name LIKE 'avatar_' || regexp_replace(auth.jwt() ->> 'email', '[^a-zA-Z0-9]', '_', 'g') || '%'
);

-- 3. Policy for UPDATE (Allows logged-in users to update only their own avatar files)
CREATE POLICY "Allow authenticated users to update their own avatars"
ON storage.objects FOR UPDATE
TO authenticated
USING (
  bucket_id = 'avatars' 
  AND name LIKE 'avatar_' || regexp_replace(auth.jwt() ->> 'email', '[^a-zA-Z0-9]', '_', 'g') || '%'
)
WITH CHECK (
  bucket_id = 'avatars' 
  AND name LIKE 'avatar_' || regexp_replace(auth.jwt() ->> 'email', '[^a-zA-Z0-9]', '_', 'g') || '%'
);

-- 4. Policy for DELETE (Allows logged-in users to delete only their own avatar files)
CREATE POLICY "Allow authenticated users to delete their own avatars"
ON storage.objects FOR DELETE
TO authenticated
USING (
  bucket_id = 'avatars' 
  AND name LIKE 'avatar_' || regexp_replace(auth.jwt() ->> 'email', '[^a-zA-Z0-9]', '_', 'g') || '%'
);

-- ===============================================================================
-- Bank deposit ledger: records money moved from collected cash into the bank account so the
-- Bank Deposits screen can derive Cash in Hand (collected minus banked) (Version: 1.0.29)
-- ===============================================================================
CREATE TABLE IF NOT EXISTS public.bank_deposits (
    id SERIAL PRIMARY KEY,
    amount DOUBLE PRECISION NOT NULL,
    date_text TEXT NOT NULL,
    timestamp BIGINT NOT NULL,
    deposited_by_id INTEGER REFERENCES public.members(id) ON DELETE SET NULL,
    deposited_by_name TEXT NOT NULL,
    sync_key TEXT NOT NULL DEFAULT (gen_random_uuid())::text UNIQUE,
    deleted BOOLEAN DEFAULT false,
    deleted_at TIMESTAMP WITH TIME ZONE,
    user_id UUID REFERENCES auth.users(id) ON DELETE CASCADE DEFAULT auth.uid(),
    created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- Comment: ON DELETE SET NULL (not CASCADE) so removing a member never erases a financial record;
-- the denormalized deposited_by_name keeps the ledger row readable afterwards.
CREATE INDEX IF NOT EXISTS idx_bank_deposits_deleted ON public.bank_deposits(deleted);

ALTER TABLE public.bank_deposits ENABLE ROW LEVEL SECURITY;

    FOR SELECT TO authenticated USING (true);

    ) WITH CHECK (

-- ===============================================================================
-- Per-member Weekly Savings Goals: every member saves a different amount each week, so this shared
-- table holds one row per member instead of one society-wide value (Version: 1.0.36)
-- ===============================================================================
   -- app_settings is private to each authenticated user, so the shared goals need their own table: every
   -- member reads it, only admins write it. The row is keyed by the member's lowercased email, which is the
   -- identity members are matched on across devices, so a goal follows the member everywhere.
   --
   -- The table is dropped and recreated rather than altered: the old table was a singleton
   -- (PRIMARY KEY (id) with CHECK (id = 1)), which cannot hold per-member rows, and a single society-wide value
   -- carries no per-member information worth preserving. No rows are seeded here: a member with no row yet falls
   -- back to the 500৳ default, and their first row is created when an admin saves their goal.
   DROP TABLE IF EXISTS public.community_settings CASCADE;

   CREATE TABLE public.community_settings (
       -- Lowercased member email: the key members are matched on across devices, so each goal follows its member
       member_email TEXT PRIMARY KEY,
       weekly_goal DOUBLE PRECISION NOT NULL DEFAULT 500.0,
       -- Bumped by the trigger below on every update. Clients send the version they last read as a precondition,
       -- so an admin device still holding an older goal cannot silently overwrite a newer one set elsewhere.
       version INTEGER NOT NULL DEFAULT 1,
       -- Audit trail only: which admin last changed the goal. Deliberately not referenced by any RLS policy.
       updated_by UUID DEFAULT auth.uid(),
       created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL,
       updated_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
   );

   ALTER TABLE public.community_settings ENABLE ROW LEVEL SECURITY;

   -- PostgreSQL has no CREATE POLICY IF NOT EXISTS, so drop first: this keeps the whole script re-runnable
   -- (re-running otherwise fails with 42710 "policy already exists").
   DROP POLICY IF EXISTS "Allow read for all authenticated users" ON public.community_settings;
   -- Every authenticated member can read the shared goals so all projections match
   CREATE POLICY "Allow read for all authenticated users" ON public.community_settings
       FOR SELECT TO authenticated USING (true);

   -- Only admins may change any member's goal
   DROP POLICY IF EXISTS "Admins can manage community settings" ON public.community_settings;
   CREATE POLICY "Admins can manage community settings" ON public.community_settings
       FOR ALL TO authenticated USING (
           EXISTS (
               SELECT 1 FROM public.members
               WHERE user_id = auth.uid() AND role = 'Admin'
           )
       ) WITH CHECK (
           EXISTS (
               SELECT 1 FROM public.members
               WHERE user_id = auth.uid() AND role = 'Admin'
           )
       );

   -- Server-side bookkeeping for the compare-and-set above: every update bumps the version and refreshes the audit
   -- stamp, so clients can never supply (or forge) either value themselves.
   CREATE OR REPLACE FUNCTION public.touch_community_settings()
   RETURNS trigger
   LANGUAGE plpgsql
   SECURITY INVOKER

SET search_path = public
AS $$
BEGIN
    NEW.updated_at := timezone('utc'::text, now());
    NEW.updated_by := auth.uid();
    NEW.version := OLD.version + 1;
    RETURN NEW;
END;
$$;

-- No EXECUTE grant is revoked here: the trigger function runs as the writing user, so revoking would fail the
-- update itself, and the function is inert as a /rpc target because trigger functions can only run as triggers.
DROP TRIGGER IF EXISTS community_settings_touch ON public.community_settings;
CREATE TRIGGER community_settings_touch
    BEFORE UPDATE ON public.community_settings
    FOR EACH ROW
    EXECUTE FUNCTION public.touch_community_settings();
